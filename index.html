SQL Analysis Queries

1. Order Details with Customer & Product


SELECT
    o.OrderID,
    o.OrderDate,
    c.CustomerName,
    c.Region,
    p.ProductName,
    p.Category,
    o.Quantity,
    o.Sales,

    CASE
        WHEN o.Sales >= 10000
        THEN 'High Value'
        ELSE 'Regular'
    END AS OrderStatus,

    ROUND(o.Sales / o.Quantity, 2)
    AS Unit_Selling_Price

FROM Orders o

INNER JOIN Customers c
    ON o.CustomerID = c.CustomerID

INNER JOIN Products p
    ON o.ProductID = p.ProductID;

2. Overall Sales Summary


SELECT
    SUM(Sales) AS Total_Sales,
    COUNT(*) AS Total_Orders,
    ROUND(AVG(Sales), 2) AS Average_Order_Value,
    MAX(Sales) AS Highest_Order_Value
FROM Orders;

3. Region & Category Wise Sales


SELECT
    c.Region,
    p.Category,
    SUM(o.Sales) AS Total_Sales

FROM Orders o

JOIN Customers c
    ON o.CustomerID = c.CustomerID

JOIN Products p
    ON o.ProductID = p.ProductID

GROUP BY
    c.Region,
    p.Category

ORDER BY
    Total_Sales DESC;

4. Monthly Sales


SELECT
    DATE_FORMAT(OrderDate, '%Y-%m') AS Month,
    SUM(Sales) AS Monthly_Sales

FROM Orders

GROUP BY
    DATE_FORMAT(OrderDate, '%Y-%m');

5. Product Ranking


SELECT
    p.ProductName,
    SUM(o.Sales) AS Total_Sales,

    RANK() OVER (
        ORDER BY SUM(o.Sales) DESC
    ) AS Product_Rank

FROM Orders o

JOIN Products p
    ON o.ProductID = p.ProductID

GROUP BY
    p.ProductName

ORDER BY
    Product_Rank;

6. Top 5 Products


SELECT
    p.ProductName,
    SUM(o.Sales) AS Total_Sales

FROM Orders o

JOIN Products p
    ON o.ProductID = p.ProductID

GROUP BY
    p.ProductName

ORDER BY
    Total_Sales DESC

LIMIT 5;
